import java.util.Scanner;

public class BullyAlgorithm {
    static int n;
    static int[] pro = new int[10];
    static int[] sta = new int[10];
    static int co;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        System.out.print("Enter the number of processes: ");
        n = in.nextInt();

        for (int i = 0; i < n; i++) {
            System.out.println("For Process " + (i + 1) + ":");
            System.out.print("---> Status (1 = alive, 0 = down): ");
            sta[i] = in.nextInt();
            System.out.print("---> Priority: ");
            pro[i] = in.nextInt();
        }

        System.out.print("Which process will initiate election? (1 to " + n + "): ");
        int ele = in.nextInt();
        elect(ele);

        System.out.println("Final coordinator is Process " + co);
    }

    static void elect(int ele) {
        ele = ele - 1; // Convert to 0-based index
        for (int i = 0; i < n; i++) {
            if (pro[ele] < pro[i] && sta[i] == 1) {
                System.out.println("Election message sent from Process " + (ele + 1) + " to Process " + (i + 1));
                elect(i + 1); // Recursive call for higher process
            }
        }

        // After election, find the highest-priority alive process as coordinator
        int max = -1;
        for (int i = 0; i < n; i++) {
            if (sta[i] == 1 && pro[i] > max) {
                max = pro[i];
                co = i + 1;
            }
        }
    }
}




import java.util.Scanner;

public class RingAlgorithm {
    static int n;
    static int[] pro = new int[100]; // Priorities
    static int[] sta = new int[100]; // Status (1 = alive, 0 = down)
    static int co; // Coordinator

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        System.out.print("Enter the number of processes: ");
        n = in.nextInt();

        for (int i = 0; i < n; i++) {
            System.out.println("For Process " + (i + 1) + ":");
            System.out.print("---> Status (1 = alive, 0 = down): ");
            sta[i] = in.nextInt();
            System.out.print("---> Priority: ");
            pro[i] = in.nextInt();
        }

        System.out.println("The ring is formed as:");
        for (int i = 1; i <= n; i++) {
            System.out.print(i + " -> ");
        }
        System.out.println("1");

        System.out.print("Which process will initiate election? (1 to " + n + "): ");
        int ele = in.nextInt();
        elect(ele);

        System.out.println("Final coordinator is Process " + co);
    }

    static void elect(int ele) {
        int start = ele - 1; // Convert to 0-based index
        int i = start;
        int maxPriority = -1;
        int coordinator = -1;

        do {
            if (sta[i] == 1) {
                System.out.println("Election message passed to Process " + (i + 1) + " (Priority: " + pro[i] + ")");
                if (pro[i] > maxPriority) {
                    maxPriority = pro[i];
                    coordinator = i + 1;
                }
            }
            i = (i + 1) % n; // Next in ring
        } while (i != start);

        co = coordinator;
    }
}
